------------------ INSERT ----------------------

   (single record insert karta hai)

   {id:2, name:"Shyam", company:"Infosys", salary:50000},

   (multiple records insert karta hai)

------------------------------------------------

------------------ SELECT ----------------------

1. db.employee.find()
   (all records show karta hai)

2. db.employee.find({name:"Ram"})
   (exact search)

3. db.employee.find({salary:{$gt:50000}})
   (salary > 50000)

4. db.employee.find({salary:{$gte:50000}})
   (salary >= 50000)

------------------------------------------------

------------------ SEARCHING ------------------

1. db.employee.find({name:/^R/})
   (name starts with R)

2. db.employee.find({name:/m$/})
   (name ends with m)

3. db.employee.find({name:/a/})
   (name contains 'a')

------------------------------------------------

------------------ SORTING ------------------

1. db.employee.find().sort({salary:1})
   (ascending order)

2. db.employee.find().sort({salary:-1})
   (descending order)

------------------------------------------------

------------------ PAGINATION ------------------

1. db.employee.find().limit(5)
   (first 5 records)

2. db.employee.find().skip(5).limit(5)
   (pagination – next 5 records)

------------------------------------------------

------------------ UPDATE ------------------

1. db.employee.updateOne(
   {id:1},
   {$set:{name:"Shyam"}}
)
   (single record update karta hai)

2. db.employee.updateMany(
   {company:"TCS"},
   {$set:{salary:60000}}
)
   (multiple records update karta hai)

------------------------------------------------

------------------ DELETE ------------------

1. db.employee.deleteOne({id:1})
   (single record delete karta hai)

2. db.employee.deleteMany({salary:{$lt:50000}})
   (condition based delete karta hai)

------------------------------------------------

------------------ AGGREGATE FUNCTIONS ------------------

1. db.employee.countDocuments()
   (total records count)

2. db.employee.aggregate([
   {$group:{_id:"$company", totalSalary:{$sum:"$salary"}}}
])
   (company wise total salary)

3. db.employee.aggregate([
   {$group:{_id:null, avgSalary:{$avg:"$salary"}}}
])
   (average salary)

4. db.employee.aggregate([
   {$group:{_id:null, minSalary:{$min:"$salary"}}}
])
   (minimum salary)

5. db.employee.aggregate([
   {$group:{_id:null, maxSalary:{$max:"$salary"}}}
])
   (maximum salary)

------------------------------------------------

------------------ JOIN (LOOKUP) ------------------

1. db.employee.aggregate([
   {
     $lookup:{
       from:"department",
       localField:"dept_id",
       foreignField:"id",
       as:"dept"
     }
   }
])
   (join operation – MongoDB style)
